cmake_minimum_required(VERSION 3.20)
project(pgfire VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PGFIRE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PostgreSQL REQUIRED)
find_path(JSONC_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSONC_LIBRARY json-c REQUIRED)
add_library(pgfire_store src/store.c)
target_include_directories(pgfire_store PUBLIC include ${JSONC_INCLUDE_DIR})
target_link_libraries(pgfire_store PUBLIC CURL::libcurl OpenSSL::Crypto ${JSONC_LIBRARY})
target_compile_definitions(pgfire_store PUBLIC _POSIX_C_SOURCE=200809L)
add_executable(pgfire src/main.c src/replication.c)
target_link_libraries(pgfire PRIVATE pgfire_store PostgreSQL::PostgreSQL)
foreach(target pgfire_store pgfire)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  if(PGFIRE_SANITIZE)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_executable(test_codec tests/unit/test_codec.c)
target_link_libraries(test_codec PRIVATE pgfire_store)
add_test(NAME codec COMMAND test_codec)
add_executable(test_fencing tests/unit/test_fencing.c)
target_link_libraries(test_fencing PRIVATE pgfire_store)
install(TARGETS pgfire RUNTIME DESTINATION bin)
